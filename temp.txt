def save_importance(model, feature_cols, path, title):
    fi = pd.Series(model.feature_importances_, index=feature_cols).sort_values()
    _, ax = plt.subplots(figsize=(10, max(6, len(fi) * 0.28)))
    fi.tail(30).plot.barh(ax=ax, color="#1D4ED8")
    ax.set_title(title, fontsize=13); ax.set_xlabel("Importance")
    plt.tight_layout(); path.parent.mkdir(parents=True, exist_ok=True)
    plt.savefig(path, bbox_inches="tight", dpi=120); plt.close()

def eval_row(label, y_true, y_pred):
    yt, yp = np.array(y_true, float), np.array(y_pred, float)
    mae  = float(mean_absolute_error(yt, yp))
    rmse = float(np.sqrt(mean_squared_error(yt, yp)))
    mp   = mape_score(yt, yp)
    logger.info("  %-45s  MAE=%10.1f  MAPE=%5.1f%%", label, mae, mp)
    return {"label": label, "mae": mae, "rmse": rmse, "mape": mp}


def best_per_target(h_test):
    if h_test.empty:
        return h_test
    h_sorted = h_test.sort_values("horizon")
    best = h_sorted.groupby(["series_id", "target_ym"]).first().reset_index()
    return best


# ══════════════════════════════════════════════════════════════════════════════
# Seasonal Decomposition
# ══════════════════════════════════════════════════════════════════════════════

def compute_seasonal_indices(df, target_col, sid_col="series_id"):
    df = df.copy()
    if "month_num" not in df.columns:
        df["month_num"] = pd.PeriodIndex(df["year_month"], freq="M").month
    monthly_avg = df.groupby([sid_col, "month_num"])[target_col].mean().reset_index().rename(columns={target_col: "month_avg"})
    overall_avg = df.groupby(sid_col)[target_col].mean().reset_index().rename(columns={target_col: "overall_avg"})
    si = monthly_avg.merge(overall_avg, on=sid_col)
    si["seasonal_index"] = np.where(si["overall_avg"] > 0, si["month_avg"] / si["overall_avg"], 1.0)
    si["seasonal_index"] = si["seasonal_index"].clip(0.2, 5.0)
    return si[[sid_col, "month_num", "seasonal_index"]]

# Direct Multi-Horizon Dataset Builder
# ══════════════════════════════════════════════════════════════════════════════

def build_direct_horizon_dataset(df, target_col, max_horizon=12, sid_col="series_id"):
    """Vectorised multi-horizon dataset builder — ~50× faster than the old row loop."""
    df = df.sort_values([sid_col, "year_month"]).copy().reset_index(drop=True)
    # Columns to shift to obtain target values
    target_shift_cols = ["year_month", target_col]
    if "deseas_target" in df.columns:
        target_shift_cols.append("deseas_target")
    if "seasonal_index" in df.columns:
        target_shift_cols.append("seasonal_index")

    frames = []
    for h in range(1, max_horizon + 1):
        shifted = df.groupby(sid_col, sort=False)[target_shift_cols].shift(-h)
        valid = shifted["year_month"].notna()
        if valid.sum() == 0:
            continue
        frame = df[valid].copy()
        frame["horizon"] = h
        frame["target_ym"]  = shifted.loc[valid, "year_month"].values
        frame["target_raw"] = shifted.loc[valid, target_col].values
        frame["target_deseas"] = (
            shifted.loc[valid, "deseas_target"].values
            if "deseas_target" in shifted.columns
            else shifted.loc[valid, target_col].values
        )
        frame["target_seasonal_index"] = (
            shifted.loc[valid, "seasonal_index"].values
            if "seasonal_index" in shifted.columns
            else 1.0
        )
        frame["target_month_num"] = pd.PeriodIndex(frame["target_ym"], freq="M").month
        frames.append(frame)
    return pd.concat(frames, ignore_index=True) if frames else pd.DataFrame()


# ══════════════════════════════════════════════════════════════════════════════
# LightGBM P10/P50/P90
# ══════════════════════════════════════════════════════════════════════════════

_BASE = dict(n_estimators=800, learning_rate=0.025, num_leaves=63, max_depth=7,
             min_child_samples=15, subsample=0.80, colsample_bytree=0.75,
             reg_alpha=0.3, reg_lambda=1.5, random_state=42, n_jobs=-1, verbose=-1)

def _lgb_params(q=None):
    if q is None: return {**_BASE, "objective": "regression_l1", "metric": "mae"}
    return {**_BASE, "objective": "quantile", "metric": "quantile", "alpha": q, "n_estimators": 600}

def train_triple(X_tr, y_tr, X_val, y_val, sw=None):
    span = get_current_span()
    results = {}
    for qn, q in [("p10", 0.10), ("p50", None), ("p90", 0.90)]:
        span.add_event("lgb_fit_start", {"quantile": qn, "train_rows": len(X_tr), "val_rows": len(X_val), "features": len(X_tr.columns)})
        t_fit = time.time()
        logger.info("    [%s] fitting  train=%s  val=%s  features=%s",
                    qn.upper(), f"{len(X_tr):,}", f"{len(X_val):,}", len(X_tr.columns))
        m = lgb.LGBMRegressor(**_lgb_params(q))
        cb = [lgb.early_stopping(80, verbose=False), lgb.log_evaluation(-1)]
        kw = dict(eval_set=[(X_val, y_val)], callbacks=cb)
        if sw is not None and qn == "p50": kw["sample_weight"] = sw
        m.fit(X_tr, y_tr, **kw); results[qn] = m
        bi = getattr(m, 'best_iteration_', m.n_estimators)
        elapsed = round(time.time() - t_fit, 1)
        span.add_event("lgb_fit_done", {"quantile": qn, "best_iteration": bi, "elapsed_s": elapsed})
        logger.info("    [%s] done  best_iter=%s  (%.1fs)", qn.upper(), bi, elapsed)
    return results["p10"], results["p50"], results["p90"]

def predict_triple(models, X):
    m10, m50, m90 = models
    p50 = np.maximum(m50.predict(X), 0)
    p10 = np.minimum(np.maximum(m10.predict(X), 0), p50)
    p90 = np.maximum(m90.predict(X), p50)
    return p10, p50, p90

def _get_lag_suffix(target_col):
    """Return '' for milk_kg, '_fat' for fat_kg, '_snf' for snf_kg."""
    return "" if target_col == "milk_kg" else f"_{target_col.replace('_kg', '')}"

def _build_feat_cols(horizon_df, target_col, has_sid_cat=False):
    """Build feature column list for a given target."""
    suffix = _get_lag_suffix(target_col)

    feat_cols = ["horizon", "target_month_num", "target_seasonal_index", "month_num"]
    if has_sid_cat:
        feat_cols.append("series_id_cat")

    # Target-specific lags.
    # lag_3 DROPPED per report §7 (r≈0.089, uncorrelated, PACF confirms no AR(3))
    # yoy_growth dropped for fat_kg / snf_kg: lag_12 reaches into the pre-tracking
    # zero-filled period, making yoy_growth a corrupted signal for those targets.
    # milk_kg has clean history throughout so yoy_growth remains valid there.
    lag_candidates = [f"lag_1{suffix}", f"lag_6{suffix}", f"lag_12{suffix}",
                      f"roll3{suffix}", f"roll6{suffix}", f"roll12{suffix}"]
    if target_col == "milk_kg":
        lag_candidates.append(f"yoy_growth{suffix}")
    for c in lag_candidates:
        if c in horizon_df.columns:
            feat_cols.append(c)

    # ── Cross-target lags (report §6.2: lag_12_fat is top Pearson feature for
    #    both Milk AND Fat; lag_6_fat is strong cross-signal for SNF).
    # Milk model uses fat lags and vice versa; SNF uses fat lags.
    cross_lags = []
    if target_col == "milk_kg":
        cross_lags = ["lag_12_fat", "rolling_3_mean_fat"]
    elif target_col == "fat_kg":
        cross_lags = ["lag_12", "rolling_3_mean"]  # milk lags
    elif target_col == "snf_kg":
        cross_lags = ["lag_6_fat", "lag_12_fat", "rolling_3_mean_fat"]
    for c in cross_lags:
        if c in horizon_df.columns:
            feat_cols.append(c)

    # Hist stats
    hist_mean_col = "overall_hist_mean" if not has_sid_cat else "society_hist_mean"
    hist_std_col  = "overall_hist_std"  if not has_sid_cat else "society_hist_std"
    for c in [hist_mean_col, hist_std_col]:
        if c in horizon_df.columns:
            feat_cols.append(c)

    if has_sid_cat and "district_share_lag1" in horizon_df.columns:
        feat_cols.append("district_share_lag1")

    # Calendar & weather & operational & vet visits.
    # Dropped per report §7: navratri_month, holi_month, summer_fodder_stress,
    # extreme_heat_days, society_age_months (zero-variance).
    for c in ["quarter","months_since_start","month_sin","month_cos",
              "diwali_month","fmd_vaccination_month","hs_vaccination_month",
              "brucellosis_vaccination_month","calving_season","kharif_harvest","rabi_sowing",
              "uttarayan","Q1_flag","total_holidays_in_month",
              "is_monsoon","is_summer","is_winter",
              "lumpy_skin_outbreak","post_lsd_recovery","is_lsd_period",
              "avg_temp_max","avg_temp_min","total_rainfall_mm","dry_days",
              "max_wind_speed","temp_range","rainfall_lag1",
              "avg_daily_milk","peak_day_milk","active_days","milk_concentration",
              "avg_daily_fat","peak_day_fat","avg_daily_snf","peak_day_snf",
              "unique_societies","total_transactions","cow_share",
              "society_volume_tier_enc",
              "vet_visits_lag2","vet_visits_lag6","vet_visits_rolling3","vet_visits_yoy"]:
        if c in horizon_df.columns: feat_cols.append(c)

    return list(dict.fromkeys(c for c in feat_cols if c in horizon_df.columns))

def build_overall_model(panel, target_col):
    span = get_current_span()
    suffix = _get_lag_suffix(target_col)
    tname = TARGET_DISPLAY.get(target_col, target_col)
    train_start = TRAIN_START_BY_TARGET.get(target_col, OVERALL_TRAIN_START)
    span.add_event("model_a_start", {"target": target_col, "train_start": train_start, "train_end": TRAIN_END})
    t_model = time.time()
    logger.info("MODEL A — Overall — %s", tname)
    if target_col == "snf_kg":
        logger.info("Train %s->%s  (SNF cold-start + ramp-up excluded)", train_start, TRAIN_END)
    else:
        logger.info("Train %s->%s  (2022 excluded: LSD+ramp-up)", train_start, TRAIN_END)

    df = panel[panel["level"] == "overall"].copy().sort_values(["series_id", "year_month"])
    df["month_num"] = pd.PeriodIndex(df["year_month"], freq="M").month

    logger.info("Data months: %s -> %s (%s rows)", df['year_month'].min(), df['year_month'].max(), len(df))

    # Seasonal indices and hist-stats computed on the clean target-specific window only
    span.add_event("seasonal_indices_start", {"target": target_col})
    train_only = df[(df["year_month"] >= train_start) & (df["year_month"] <= TRAIN_END)]
    seasonal_df = compute_seasonal_indices(train_only, target_col, "series_id")
    span.add_event("seasonal_indices_done")
    df = df.merge(seasonal_df, on=["series_id", "month_num"], how="left")
    df["seasonal_index"] = df["seasonal_index"].fillna(1.0)
    df["deseas_target"] = np.where(df["seasonal_index"] > 0, df[target_col] / df["seasonal_index"], df[target_col])

    # Lag/rolling features use FULL history so origin features are always populated
    grp = df.groupby("series_id")["deseas_target"]
    for lag in [1,3,6,12]:
        df[f"lag_{lag}{suffix}"] = grp.shift(lag).fillna(0)
    # Pre-shift into a temp column to avoid chaining shift().rolling() inside
    # transform, which triggers a SystemError in pandas 2.x (C-level bug).
    _tmp = "_shifted_deseas"
    df[_tmp] = grp.shift(1)
    df[f"roll3{suffix}"]  = df.groupby("series_id")[_tmp].transform(lambda x: x.rolling(3,  min_periods=1).mean()).fillna(0)
    df[f"roll6{suffix}"]  = df.groupby("series_id")[_tmp].transform(lambda x: x.rolling(6,  min_periods=1).mean()).fillna(0)
    df[f"roll12{suffix}"] = df.groupby("series_id")[_tmp].transform(lambda x: x.rolling(12, min_periods=3).mean()).fillna(0)
    df = df.drop(columns=[_tmp])

    # All targets have real data from 2021 — no lag_12 null-out needed.
    stats = train_only.groupby("series_id")[target_col].agg(["mean","std"]).reset_index()
    stats.columns = ["series_id", "overall_hist_mean", "overall_hist_std"]
    stats["overall_hist_std"] = stats["overall_hist_std"].fillna(0)
    df = df.merge(stats, on="series_id", how="left")

    span.add_event("horizon_dataset_start", {"target": target_col})
    t_hz = time.time()
    logger.info("Building horizon dataset...")
    horizon_df = build_direct_horizon_dataset(df[df["year_month"] <= TEST_END], target_col, max_horizon=24)
    if horizon_df.empty:
        logger.warning("Overall horizon dataset EMPTY — skipping %s", target_col)
        return None, [], pd.DataFrame(), [], pd.DataFrame(), {}
    span.add_event("horizon_dataset_done", {"rows": len(horizon_df), "elapsed_s": round(time.time() - t_hz, 2)})

    feat_cols = _build_feat_cols(horizon_df, target_col, has_sid_cat=False)

    # Train ONLY on clean-window rows so the model learns the post-LSD growth trajectory
    # (SNF uses a later start to skip the tracking ramp-up)
    h_train = horizon_df[(horizon_df["year_month"] >= train_start) &
                         (horizon_df["year_month"] <= TRAIN_END) &
                         (horizon_df["target_ym"]  <= TRAIN_END)]
    h_test  = horizon_df[(horizon_df["target_ym"] >= TEST_START) & (horizon_df["target_ym"] <= TEST_END)]

    test_targets = sorted(h_test["target_ym"].unique())
    logger.info("Test targets found (any horizon): %s", test_targets)

    atm = sorted(h_train["year_month"].unique())
    vc = atm[max(1, int(len(atm) * 0.8))]
    h_fit = h_train[h_train["year_month"] < vc]
    h_val = h_train[h_train["year_month"] >= vc]
    X_tr  = h_fit[feat_cols].fillna(0); y_tr = h_fit["target_deseas"]
    X_val = h_val[feat_cols].fillna(0); y_val = h_val["target_deseas"]

    logger.info("Fit: %s  Val: %s  Test: %s  Features: %s",
                f"{len(h_fit):,}", f"{len(h_val):,}", f"{len(h_test):,}", len(feat_cols))

    # LSD months are normalized in 03_aggregate.py — no sample-weight reduction needed.
    span.add_event("train_triple_start", {"model": "A_overall", "target": target_col, "fit_rows": len(h_fit), "val_rows": len(h_val), "features": len(feat_cols)})
    t_train = time.time()
    logger.info("Training P10/P50/P90 for %s...", tname)
    m10, m50, m90 = train_triple(X_tr, y_tr, X_val, y_val)
    span.add_event("train_triple_done", {"model": "A_overall", "target": target_col, "elapsed_s": round(time.time() - t_train, 2)})
    save_importance(m50, feat_cols, OUT_DIR / f"feature_importance_overall_{target_col}.png",
                    f"Feature Importance — Model A Overall ({tname})")

    eval_rows = []; test_pred_df = pd.DataFrame()
    if not h_test.empty:
        X_te = h_test[feat_cols].fillna(0)
        p10_d, p50_d, p90_d = predict_triple((m10, m50, m90), X_te)
        si = h_test["target_seasonal_index"].values
        h_test = h_test.copy()
        h_test["p10"] = np.round(np.maximum(p10_d * si, 0), 2)
        h_test["p50"] = np.round(np.maximum(p50_d * si, 0), 2)
        h_test["p90"] = np.round(np.maximum(p90_d * si, 0), 2)

        best = best_per_target(h_test)
        if best["target_raw"].sum() > 0:
            eval_rows.append(eval_row(f"overall ({target_col})", best["target_raw"], best["p50"]))

        test_pred_df = best[["series_id","target_ym","target_raw","p10","p50","p90"]].copy()
        test_pred_df = test_pred_df.rename(columns={"target_ym":"year_month","target_raw":target_col})

    MODEL_DIR.mkdir(parents=True, exist_ok=True)
    # Origin rows: last row per series with all lag/rolling features (for dynamic forecasting)
    origin_rows = df.sort_values("year_month").groupby("series_id").last().reset_index()
    with open(MODEL_DIR / f"{_VER_TAG}_lgbm_overall_{target_col}.pkl", "wb") as f:
        pickle.dump({"model": m50, "model_p10": m10, "model_p90": m90,
                     "feature_cols": feat_cols, "seasonal_indices": seasonal_df,
                     "origin_rows": origin_rows,
                     "target": target_col}, f)
    span.add_event("model_a_done", {"target": target_col, "total_elapsed_s": round(time.time() - t_model, 2)})
    logger.info("Saved: %s  (%s origins)", MODEL_DIR / f'{_VER_TAG}_lgbm_overall_{target_col}.pkl', len(origin_rows))
    return (m10, m50, m90), feat_cols, df, eval_rows, test_pred_df, seasonal_df


# ══════════════════════════════════════════════════════════════════════════════
# YoY Growth — computed from training data, applied as compound trend
# ══════════════════════════════════════════════════════════════════════════════

def compute_yoy_growth(df, target_col, sid="overall", n_years=2):
    """Compute average YoY growth rate for a series from the last n_years of data.

    Returns a monthly compound growth rate (e.g. 0.004 means ~4.8% annual).
    Returns None if insufficient data — callers should substitute the
    cross-society fallback rate so every society still gets a growth trend.
    """
    sub = df[df["series_id"] == sid].sort_values("year_month") if "series_id" in df.columns else df.sort_values("year_month")
    months = sorted(sub["year_month"].unique())
    if len(months) < 13:
        return None
    yoy_rates = []
    for ym in months[-n_years * 12:]:
        cur = sub[sub["year_month"] == ym][target_col]
        prev_ym = (pd.Period(ym, freq="M") - 12).strftime("%Y-%m")
        prev = sub[sub["year_month"] == prev_ym][target_col]
        if cur.empty or prev.empty:
            continue
        c, p = float(cur.iloc[0]), float(prev.iloc[0])
        if p > 0 and c > 0:
            yoy_rates.append((c - p) / p)
    if not yoy_rates:
        return None
    annual_rate = float(np.median(yoy_rates))
    # Convert to monthly compound rate; cap between -10% and +20% annual
    annual_rate = np.clip(annual_rate, -0.10, 0.20)
    monthly_rate = (1 + annual_rate) ** (1 / 12) - 1
    return float(monthly_rate)


def apply_growth_trend(fdf, monthly_rate):
    """Apply compound growth factor per horizon to p10/p50/p90 in-place."""
    if abs(monthly_rate) < 1e-6:
        return fdf
    factors = (1 + monthly_rate) ** fdf["horizon"].values
    for q in ["p10", "p50", "p90"]:
        fdf[q] = np.round(np.maximum(fdf[q].values * factors, 0), 2)
    return fdf


def compute_monthly_yoy_rates(panel, target_col, sid="overall", start_year=2023):
    """Compute per-calendar-month YoY growth rates from start_year onwards.

    For each calendar month (1-12), collects all year-over-year pairs available
    in the panel from start_year and takes the median growth rate.

    Months that show a seasonal softness (negative or zero median rate) are NOT
    floored at an arbitrary constant.  Instead they inherit the overall growth
    baseline — the median of all months that do show positive growth.  This keeps
    every floor rate data-driven while guaranteeing YoY growth across all months.

    Excludes 2022 (start_year=2023) to avoid LSD-distorted baseline.
    """
    ov = panel[(panel["series_id"] == sid) & (panel[target_col] > 0)].copy()
    ov["year"]      = pd.PeriodIndex(ov["year_month"], freq="M").year
    ov["month_num"] = pd.PeriodIndex(ov["year_month"], freq="M").month
    ov = ov[ov["year"] >= start_year]

    # Step 1 — raw per-month median YoY rate (can be negative)
    raw_rates = {}
    for m in range(1, 13):
        sub  = ov[ov["month_num"] == m].sort_values("year")
        vals = dict(zip(sub["year"], sub[target_col]))
        yrs  = sorted(vals)
        yoy  = []
        for i in range(1, len(yrs)):
            y0, y1 = yrs[i-1], yrs[i]
            if vals[y0] > 0:
                yoy.append((vals[y1] - vals[y0]) / vals[y0])
        raw_rates[m] = float(np.median(yoy)) if yoy else None

    # Step 2 — overall growth baseline: median of months with positive observed growth
    positive = [r for r in raw_rates.values() if r is not None and r > 0]
    baseline = float(np.median(positive)) if positive else 0.05
    logger.info("  [%s %s] YoY baseline (median of positive months): %+.2f%%",
                sid, target_col, baseline * 100)

    # Step 3 — assign final rates; replace negative/missing with the baseline
    rates = {}
    for m in range(1, 13):
        r = raw_rates[m]
        if r is None or r <= 0:
            rates[m] = baseline   # data-driven fallback, not a flat constant
        else:
            rates[m] = r
        logger.info("    month %2d: raw=%s  final=%+.2f%%",
                    m,
                    f"{r*100:+.2f}%" if r is not None else "n/a",
                    rates[m] * 100)
    return rates


def _apply_mom_floor(fdf, panel, target_col, max_scale=1.05, n_years=2):
    """Month-on-Month floor applied after the YoY floor.

    For each forecast month M, computes the median ratio actual[M]/actual[M-1]
    over the last n_years from the panel (capped at TRAIN_END so OOS actuals
    never influence the ratio).  If the current P50 is below
    prev_month_P50 × ratio, lifts P50 (and P10/P90 proportionally), capped at
    max_scale × current_P50 to prevent a single outlier prior month from pulling
    the forecast too high.

    For the first forecast month (i=0), the last actual training month (TRAIN_END)
    is used as the previous-month anchor instead of skipping the floor entirely.

    Applied sequentially so each month uses the already-lifted previous P50.
    Only runs on the overall series (society-level MoM ratios are too noisy).
    """
    fdf = fdf.copy().reset_index(drop=True)
    # Cap at TRAIN_END so April–June 2026 actuals don't influence ratio computation
    ov = panel[
        (panel["series_id"] == "overall") &
        (panel[target_col] > 0) &
        (panel["year_month"] <= TRAIN_END)
    ].copy()
    ov["year"]  = pd.PeriodIndex(ov["year_month"], freq="M").year
    ov["month"] = pd.PeriodIndex(ov["year_month"], freq="M").month
    cutoff_year = ov["year"].max() - n_years

    def _ratio(m):
        prev_m = 12 if m == 1 else m - 1
        ratios = []
        for yr in ov["year"].unique():
            if yr <= cutoff_year:
                continue
            prev_yr = yr - 1 if m == 1 else yr
            cur  = ov[(ov["year"] == yr)    & (ov["month"] == m)][target_col]
            prev = ov[(ov["year"] == prev_yr) & (ov["month"] == prev_m)][target_col]
            if not cur.empty and not prev.empty and float(prev.iloc[0]) > 0:
                ratios.append(float(cur.iloc[0]) / float(prev.iloc[0]))
        return float(np.median(ratios)) if ratios else None

    # Last actual value from training data — used as anchor for the first forecast month
    ov_sorted = ov.sort_values("year_month")
    last_actual_p50 = float(ov_sorted[target_col].iloc[-1]) if not ov_sorted.empty else None

    lifted = 0
    for i in range(len(fdf)):
        m = pd.Period(fdf.loc[i, "year_month"], freq="M").month
        ratio = _ratio(m)
        if ratio is None:
            continue
        if i == 0:
            # First forecast month: anchor on last actual (TRAIN_END month) instead of skipping
            if last_actual_p50 is None or last_actual_p50 <= 0:
                continue
            prev_p50 = last_actual_p50
        else:
            prev_p50 = fdf.loc[i - 1, "p50"]
        mom_floor = prev_p50 * ratio
        cur_p50   = fdf.loc[i, "p50"]
        if cur_p50 > 0 and cur_p50 < mom_floor:
            new_p50 = min(mom_floor, cur_p50 * max_scale)
            scale   = new_p50 / cur_p50
            for q in ["p10", "p50", "p90"]:
                fdf.loc[i, q] = round(fdf.loc[i, q] * scale, 2)
            lifted += 1
    if lifted:
        logger.info("  [MoM floor] %d month(s) lifted (max_scale=%.0f%%)", lifted, max_scale * 100)
    return fdf


def _apply_yoy_floor(fdf, hist_df, target_col, monthly_rates,
                     dead_band=0.0, max_scale=None, label=""):
    """YoY floor for a forecast DataFrame.

    Compares each forecast month against the same month last year in hist_df.
    Floor = last_year_actual × (1 + per_month_rate).
    Lifts P10/P50/P90 proportionally when P50 falls below the floor.

    dead_band : fractional tolerance before floor triggers (0.0 = original behaviour,
                fire as soon as p50 < floor; 0.05 = only fire if 5%+ below floor)
    max_scale : cap on how much to lift (None = no cap, scale exactly to floor)
    """
    fdf = fdf.copy()
    ly_lookup    = hist_df.set_index("year_month")[target_col].to_dict()
    ly_ym_list   = [(pd.Period(ym, freq="M") - 12).strftime("%Y-%m") for ym in fdf["year_month"]]
    fdf["_ly"]   = [ly_lookup.get(ym, None) for ym in ly_ym_list]
    fdf["_mnum"] = pd.PeriodIndex(fdf["year_month"], freq="M").month
    fdf["_frate"]= fdf["_mnum"].map(monthly_rates).fillna(0.01)
    has_ly       = fdf["_ly"].notna() & (fdf["_ly"] > 0)

    floor   = fdf["_ly"] * (1 + fdf["_frate"])
    trigger = floor * (1 - dead_band)
    below   = has_ly & (fdf["p50"] < trigger) & (fdf["p50"] > 0)

    if below.any():
        scale = floor[below] / fdf.loc[below, "p50"]
        if max_scale is not None:
            scale = scale.clip(upper=max_scale)
        for q in ["p10", "p50", "p90"]:
            fdf.loc[below, q] = np.round(fdf.loc[below, q] * scale, 2)
        logger.info("  [YoY floor%s] %d month(s) lifted to per-month growth floor",
                    f" {label}" if label else "", below.sum())

    fdf["yoy_growth_pct"] = None
    if has_ly.any():
        fdf.loc[has_ly, "yoy_growth_pct"] = (
            (fdf.loc[has_ly, "p50"] / fdf.loc[has_ly, "_ly"] - 1) * 100
        ).round(2)

    return fdf.drop(columns=["_ly", "_mnum", "_frate"])

# ══════════════════════════════════════════════════════════════════════════════

def forecast_overall(models, feat_cols, df_all, panel, seasonal_df, target_col):
    suffix = _get_lag_suffix(target_col)
    tname = TARGET_DISPLAY.get(target_col, target_col)
    logger.info("Forecasting overall %s...", tname)
    if models is None or df_all.empty:
        return pd.DataFrame(columns=["series_id","year_month","p10","p50","p90"])
    (m10, m50, m90) = models
    origin = df_all.sort_values("year_month").iloc[-1]
    wx_by_m = _weather_by_month(panel)

    # Per-month YoY rates — exclude the last panel month (always partial).
    # SNF uses 2024+ (one clean YoY pair); Milk/Fat use 2023+ (post-LSD).
    yoy_start_year = 2024 if target_col == "snf_kg" else 2023
    _last_ym = panel[panel["series_id"] == "overall"]["year_month"].max()
    _panel_for_rates = panel[(panel["year_month"] < _last_ym) & (panel["year_month"] <= TRAIN_END)]
    monthly_rates = compute_monthly_yoy_rates(_panel_for_rates, target_col, sid="overall", start_year=yoy_start_year)
    avg_annual = np.mean([(1 + r) ** 12 - 1 for r in monthly_rates.values()])
    logger.info("  Per-month YoY rates — avg annual: %+.1f%%", avg_annual * 100)

    vv_by_m = _vet_visits_by_month(panel)

    # Autoregressive value lookup — seed with historical actuals plus origin's
    # own target value so that lag_1 for forecast month 1 = last actual (M),
    # not origin's lag_1 which is M-1.
    ov_vals = (
        panel[
            (panel["series_id"] == "overall") &
            (panel[target_col] > 0) &
            (panel["year_month"] <= TRAIN_END)
        ]
        .sort_values("year_month")[["year_month", target_col]]
    )
    ar_vals = ov_vals.set_index("year_month")[target_col].to_dict()
    origin_ym = str(origin.get("year_month", ""))
    if origin_ym and origin_ym not in ar_vals:
        ar_vals[origin_ym] = safe(origin.get(target_col, 0))

    def _ar_lag(ym_str, n):
        lag_ym = (pd.Period(ym_str, freq="M") - n).strftime("%Y-%m")
        return ar_vals.get(lag_ym, None)

    rows = []
    for h_idx, ym in enumerate(FORECAST_MONTHS[:24], 1):
        p = pd.Period(ym, freq="M"); tm = p.month
        cal = calendar_flags(p.year, tm); wx = wx_by_m.get(tm, {}); vv = vv_by_m.get(tm, {})
        si_row = seasonal_df[(seasonal_df["series_id"] == "overall") & (seasonal_df["month_num"] == tm)]
        si_val = float(si_row["seasonal_index"].iloc[0]) if not si_row.empty else 1.0
        row = {"series_id": "overall", "year_month": ym, "horizon": h_idx,
               "target_month_num": tm, "target_seasonal_index": si_val,
               "month_num": origin.get("month_num", 0)}

        # Autoregressive lag features: look up from ar_vals (actuals + prior predictions)
        l1  = _ar_lag(ym, 1)
        l6  = _ar_lag(ym, 6)
        l12 = _ar_lag(ym, 12)
        row[f"lag_1{suffix}"]  = safe(l1)  if l1  is not None else safe(origin.get(f"lag_1{suffix}",  0))
        row[f"lag_6{suffix}"]  = safe(l6)  if l6  is not None else safe(origin.get(f"lag_6{suffix}",  0))
        row[f"lag_12{suffix}"] = safe(l12) if l12 is not None else safe(origin.get(f"lag_12{suffix}", 0))

        # Rolling means: recompute from ar_vals so months 13-24 use the updated
        # AR prediction history rather than the frozen March-2026 origin values.
        # ar_vals holds raw (seasonally multiplied) values; divide by SI to get
        # the deseasoned value the model was trained on.
        def _ar_roll(n, fallback_key):
            deseas = []
            for i in range(1, n + 1):
                lag_p   = p - i
                lag_ym  = lag_p.strftime("%Y-%m")
                raw_v   = ar_vals.get(lag_ym)
                if raw_v is None:
                    break
                si_r = seasonal_df[
                    (seasonal_df["series_id"] == "overall") &
                    (seasonal_df["month_num"]  == lag_p.month)
                ]
                si_l = float(si_r["seasonal_index"].iloc[0]) if not si_r.empty else 1.0
                deseas.append(raw_v / si_l if si_l > 0 else raw_v)
            return safe(np.mean(deseas)) if deseas else safe(origin.get(fallback_key, 0))

        row[f"roll3{suffix}"]  = _ar_roll(3,  f"roll3{suffix}")
        row[f"roll6{suffix}"]  = _ar_roll(6,  f"roll6{suffix}")
        row[f"roll12{suffix}"] = _ar_roll(12, f"roll12{suffix}")

        l1v  = safe(l1)  if l1  is not None else 0
        l12v = safe(l12) if l12 is not None else 0
        row[f"yoy_growth{suffix}"] = (
            (l1v - l12v) / l12v * 100
            if l12v > 0 else safe(origin.get(f"yoy_growth{suffix}", 0))
        )
        for k in ["overall_hist_mean", "overall_hist_std"]:
            row[k] = safe(origin.get(k, 0))
        for k in ["lag_12_fat","rolling_3_mean_fat","lag_6_fat",
                  "lag_12","rolling_3_mean","lag_12_snf","rolling_3_mean_snf"]:
            if k in origin.index:
                row[k] = safe(origin.get(k, 0))
        for c in ["avg_daily_milk","peak_day_milk","active_days","milk_concentration",
                   "avg_daily_fat","peak_day_fat","avg_daily_snf","peak_day_snf",
                   "unique_societies","total_transactions","cow_share"]:
            if c in origin.index: row[c] = safe(origin[c])
        row.update(cal); row.update(wx); row.update(vv)

        logger.info("  AR rollout %2d/24 — %s  (lag_1=%.0f  lag_12=%.0f  roll3=%.0f)",
                    h_idx, ym,
                    row.get(f"lag_1{suffix}", 0),
                    row.get(f"lag_12{suffix}", 0),
                    row.get(f"roll3{suffix}", 0))

        # Predict this month, then store P50 so future months can use it as a lag
        fdf_row = pd.DataFrame([row])
        for c in feat_cols:
            if c not in fdf_row.columns: fdf_row[c] = 0
        X_row = fdf_row[feat_cols].fillna(0)
        p10_r, p50_r, p90_r = predict_triple((m10, m50, m90), X_row)
        si_arr = np.array([si_val])
        row["p10"] = float(np.round(np.maximum(p10_r * si_arr, 0), 2)[0])
        row["p50"] = float(np.round(np.maximum(p50_r * si_arr, 0), 2)[0])
        row["p90"] = float(np.round(np.maximum(p90_r * si_arr, 0), 2)[0])
        ar_vals[ym] = row["p50"]
        rows.append(row)

    fdf = pd.DataFrame(rows)

    # ── Tactical YoY floor (overall) ─────────────────────────────────────────
    ov_hist = panel[(panel["series_id"] == "overall") & (panel[target_col] > 0)][["year_month", target_col]]
    _year1_ref = fdf[["year_month", "p50"]].rename(columns={"p50": target_col})
    # The first forecast month equals the last panel month, which is a partial month
    # (data collection was still running when training ran). Its actual is artificially
    # low, making it a bad floor anchor for the same month next year.  Remove it from
    # actuals so the forecast P50 (a full-month estimate) is used instead.
    ov_hist_adj = ov_hist[ov_hist["year_month"] != FORECAST_MONTHS[0]]
    ov_hist_ext = pd.concat([ov_hist_adj, _year1_ref], ignore_index=True).drop_duplicates(subset=["year_month"], keep="first")
    fdf = _apply_yoy_floor(fdf, ov_hist_ext, target_col, monthly_rates,
                           dead_band=0.0, max_scale=None, label="overall-pass1")
    # Pass 2: rebuild reference using floored year-1 P50 values so year-2 months
    # (e.g. Jan-2028) are floored against the already-lifted Jan-2027, not the
    # raw pre-floor Jan-2027 that _year1_ref captured above.
    _year1_ref_floored = fdf[["year_month", "p50"]].rename(columns={"p50": target_col})
    ov_hist_ext2 = pd.concat([ov_hist_adj, _year1_ref_floored], ignore_index=True).drop_duplicates(subset=["year_month"], keep="first")
    fdf = _apply_yoy_floor(fdf, ov_hist_ext2, target_col, monthly_rates,
                           dead_band=0.0, max_scale=None, label="overall-pass2")

    # Month-on-Month floor: lifts months where model underpredicts seasonal
    # transition (e.g. May after a strong April), capped at 5% per month.
    fdf = _apply_mom_floor(fdf, panel, target_col, max_scale=1.05, n_years=2)

    # Log per-month table (after floor adjustment)
    ly_lookup_log = ov_hist.set_index("year_month")[target_col].to_dict()
    logger.info("%s — Month-by-Month Forecast vs Last Year (after YoY floor)", tname)
    logger.info("%-10s %12s %12s %12s %9s", "Month", "P50 Fcst", "P90 Fcst", "Last Yr", "YoY(P50)")
    for _, row in fdf.iterrows():
        ly_ym = (pd.Period(row["year_month"], freq="M") - 12).strftime("%Y-%m")
        ly    = ly_lookup_log.get(ly_ym)
        yoy   = row.get("yoy_growth_pct")
        ly_s  = f"{float(ly):>12,.0f}"  if ly  is not None      else f"{'N/A':>12}"
        yoy_s = f"{float(yoy):>+8.1f}%" if pd.notna(yoy) else f"{'N/A':>9}"
        logger.info("%-10s %12s %12s %s %s",
                    row["year_month"], f"{row['p50']:,.0f}", f"{row['p90']:,.0f}", ly_s, yoy_s)

    return fdf[["series_id","year_month","p10","p50","p90","yoy_growth_pct"]]

def build_society_model(panel, target_col):
    span = get_current_span()
    suffix = _get_lag_suffix(target_col)
    tname = TARGET_DISPLAY.get(target_col, target_col)
    train_start = TRAIN_START_BY_TARGET.get(target_col, OVERALL_TRAIN_START)
    span.add_event("model_b_start", {"target": target_col, "train_start": train_start})
    t_model = time.time()
    logger.info("MODEL B — Society — %s", tname)
    logger.info("Train %s->%s  Test %s->%s", train_start, TRAIN_END, TEST_START, TEST_END)

    df = panel[panel["level"] == "society"].copy().sort_values(["series_id", "year_month"])
    # Exclude __OTHERS__ — it's a residual, not a modelable series
    df = df[df["series_id"] != OTHERS_SERIES_ID].copy()
    df["month_num"] = pd.PeriodIndex(df["year_month"], freq="M").month

    logger.info("Societies: %s  Data: %s -> %s",
                df['series_id'].nunique(), df['year_month'].min(), df['year_month'].max())

    # Apply per-target train start window (SNF excludes cold-start + ramp-up)
    train_only = df[(df["year_month"] >= train_start) & (df["year_month"] <= TRAIN_END)]
    seasonal_df = compute_seasonal_indices(train_only, target_col, "series_id")
    df = df.merge(seasonal_df, on=["series_id", "month_num"], how="left")
    df["seasonal_index"] = df["seasonal_index"].fillna(1.0)
    df["deseas_target"] = np.where(df["seasonal_index"] > 0, df[target_col] / df["seasonal_index"], df[target_col])

    grp = df.groupby("series_id")["deseas_target"]
    for lag in [1,3,6,12]:
        df[f"lag_{lag}{suffix}"] = grp.shift(lag).fillna(0)
    _tmp = "_shifted_deseas"
    df[_tmp] = grp.shift(1)
    df[f"roll3{suffix}"]  = df.groupby("series_id")[_tmp].transform(lambda x: x.rolling(3,  min_periods=1).mean()).fillna(0)
    df[f"roll6{suffix}"]  = df.groupby("series_id")[_tmp].transform(lambda x: x.rolling(6,  min_periods=1).mean()).fillna(0)
    df[f"roll12{suffix}"] = df.groupby("series_id")[_tmp].transform(lambda x: x.rolling(12, min_periods=3).mean()).fillna(0)
    df = df.drop(columns=[_tmp])

    # All targets have real data from 2021 — no lag_12 null-out needed.
    df["series_id_cat"] = df["series_id"].astype("category").cat.codes
    sid_map = dict(enumerate(df["series_id"].astype("category").cat.categories))

    stats = train_only.groupby("series_id")[target_col].agg(["mean","std"]).reset_index()
    stats.columns = ["series_id", "society_hist_mean", "society_hist_std"]
    stats["society_hist_std"] = stats["society_hist_std"].fillna(0)
    df = df.merge(stats, on="series_id", how="left")

    overall = panel[panel["series_id"] == "overall"][["year_month", target_col]].rename(columns={target_col: "district_total"})
    df = df.merge(overall, on="year_month", how="left")
    df["district_share"] = np.where(df["district_total"] > 0, df[target_col] / df["district_total"], 0)
    df["district_share_lag1"] = df.groupby("series_id")["district_share"].shift(1).fillna(0)
    df = df.drop(columns=["district_total", "district_share"])
    if "society_volume_tier" in df.columns:
        df["society_volume_tier_enc"] = df["society_volume_tier"].map(TIER_MAP).fillna(0).astype(int)

    span.add_event("horizon_dataset_start", {"model": "B_society", "target": target_col})
    t_hz = time.time()
    logger.info("Building horizon dataset...")
    horizon_df = build_direct_horizon_dataset(df[df["year_month"] <= TEST_END], target_col, max_horizon=24)
    if horizon_df.empty:
        logger.warning("Society horizon dataset EMPTY — skipping %s", target_col)
        return None, [], pd.DataFrame(), {}, [], pd.DataFrame(), {}
    span.add_event("horizon_dataset_done", {"model": "B_society", "rows": len(horizon_df), "elapsed_s": round(time.time() - t_hz, 2)})

    feat_cols = _build_feat_cols(horizon_df, target_col, has_sid_cat=True)

    h_train = horizon_df[(horizon_df["year_month"] >= train_start) &
                         (horizon_df["year_month"] <= TRAIN_END) &
                         (horizon_df["target_ym"]  <= TRAIN_END)]
    h_test  = horizon_df[(horizon_df["target_ym"] >= TEST_START) & (horizon_df["target_ym"] <= TEST_END)]

    # Cap training rows to avoid OOM on machines with limited RAM.
    # 1.5M rows × 55 features × float64 = ~0.66 GiB — well within limits.
    # Stratified by (series_id, horizon) so every society × horizon pair stays represented.
    _MAX_TRAIN_ROWS = 1_500_000
    if len(h_train) > _MAX_TRAIN_ROWS:
        h_train = (
            h_train
            .groupby(["series_id", "horizon"], group_keys=False)
            .apply(lambda g: g.sample(frac=_MAX_TRAIN_ROWS / len(h_train), random_state=42))
            .sort_values("year_month")
            .reset_index(drop=True)
        )
        logger.info("Society h_train sub-sampled to %s rows (memory cap)", f"{len(h_train):,}")

    atm = sorted(h_train["year_month"].unique())
    vc = atm[max(1, int(len(atm) * 0.8))] if atm else TRAIN_END
    h_fit = h_train[h_train["year_month"] < vc]
    h_val = h_train[h_train["year_month"] >= vc]
    X_tr  = h_fit[feat_cols].fillna(0); y_tr = h_fit["target_deseas"]
    X_val = h_val[feat_cols].fillna(0); y_val = h_val["target_deseas"]

    logger.info("Fit: %s  Val: %s  Test: %s  Features: %s",
                f"{len(h_fit):,}", f"{len(h_val):,}", f"{len(h_test):,}", len(feat_cols))

    # LSD months are normalized in 03_aggregate.py — no sample-weight reduction needed.
    span.add_event("train_triple_start", {"model": "B_society", "target": target_col, "fit_rows": len(h_fit), "features": len(feat_cols)})
    t_train = time.time()
    logger.info("Training P10/P50/P90 for %s...", tname)
    m10, m50, m90 = train_triple(X_tr, y_tr, X_val, y_val)
    span.add_event("train_triple_done", {"model": "B_society", "target": target_col, "elapsed_s": round(time.time() - t_train, 2)})
    save_importance(m50, feat_cols, OUT_DIR / f"feature_importance_society_{target_col}.png",
                    f"Feature Importance — Model B Society ({tname})")

    eval_rows = []; test_pred_df = pd.DataFrame()
    if not h_test.empty:
        X_te = h_test[feat_cols].fillna(0)
        p10_d, p50_d, p90_d = predict_triple((m10, m50, m90), X_te)
        si = h_test["target_seasonal_index"].values
        h_test = h_test.copy()
        h_test["p10"] = np.round(np.maximum(p10_d * si, 0), 2)
        h_test["p50"] = np.round(np.maximum(p50_d * si, 0), 2)
        h_test["p90"] = np.round(np.maximum(p90_d * si, 0), 2)

        best = best_per_target(h_test)
        for sid in sorted(best["series_id"].unique()):
            sub = best[best["series_id"] == sid]
            if sub["target_raw"].sum() == 0: continue
            eval_rows.append(eval_row(f"{sid} ({target_col})", sub["target_raw"], sub["p50"]))

        test_pred_df = best[["series_id","target_ym","target_raw","p10","p50","p90"]].copy()
        test_pred_df = test_pred_df.rename(columns={"target_ym":"year_month","target_raw":target_col})

    MODEL_DIR.mkdir(parents=True, exist_ok=True)
    # Origin rows: last row per society with all lag/rolling features (for dynamic forecasting)
    origin_rows = df.sort_values("year_month").groupby("series_id").last().reset_index()
    with open(MODEL_DIR / f"{_VER_TAG}_lgbm_society_{target_col}.pkl", "wb") as f:
        pickle.dump({"model": m50, "model_p10": m10, "model_p90": m90,
                     "feature_cols": feat_cols, "series_id_map": sid_map,
                     "seasonal_indices": seasonal_df, "origin_rows": origin_rows,
                     "target": target_col}, f)
    span.add_event("model_b_done", {"target": target_col, "total_elapsed_s": round(time.time() - t_model, 2)})
    logger.info("Saved: %s  (%s origins)", MODEL_DIR / f'{_VER_TAG}_lgbm_society_{target_col}.pkl', len(origin_rows))
    return (m10, m50, m90), feat_cols, df, sid_map, eval_rows, test_pred_df, seasonal_df

def forecast_society(models, feat_cols, df_all, sid_map, panel, seasonal_df, target_col):
    suffix = _get_lag_suffix(target_col)
    tname = TARGET_DISPLAY.get(target_col, target_col)
    logger.info("Forecasting society %s...", tname)
    if models is None or df_all.empty:
        return pd.DataFrame(columns=["series_id","society","year_month","p10","p50","p90"])
    (m10, m50, m90) = models
    sid_to_cat = {v: k for k, v in sid_map.items()}
    origin_rows = df_all.sort_values("year_month").groupby("series_id").last().reset_index()
    wx_by_m = _weather_by_month(panel)

    # Compute per-society YoY growth; fall back to cross-society median for sparse series.
    # Every society gets a forecast — societies without a computable own-rate use the fallback.
    all_sids = list(df_all["series_id"].unique())
    society_growth = {}   # sid -> rate or None
    all_rates = []
    for sid in all_sids:
        rate = compute_yoy_growth(df_all, target_col, sid=sid)
        society_growth[sid] = rate
        if rate is not None:
            all_rates.append(rate)
    fallback_rate = float(np.median(all_rates)) if all_rates else 0.0
    own_rate_count = sum(1 for r in society_growth.values() if r is not None)
    fallback_count = len(all_sids) - own_rate_count
    if all_rates:
        logger.info("Society %s growth — fallback (median): %+.1f%%  range: %+.1f%% .. %+.1f%%",
                    tname,
                    ((1 + fallback_rate) ** 12 - 1) * 100,
                    ((1 + min(all_rates)) ** 12 - 1) * 100,
                    ((1 + max(all_rates)) ** 12 - 1) * 100)
    else:
        logger.warning("No growth rates computable for %s — using 0%% fallback", tname)
    logger.info("Coverage: %s/%s societies have own YoY rate, %s use fallback rate (all still forecast)",
                own_rate_count, len(all_sids), fallback_count)

    # District-level monthly rates used as reference for per-society floor.
    # Per-society monthly rates are too sparse; district rates are the reliable signal.
    yoy_start_year = 2024 if target_col == "snf_kg" else 2023
    _last_ym = panel[panel["series_id"] == "overall"]["year_month"].max()
    _panel_for_rates = panel[(panel["year_month"] < _last_ym) & (panel["year_month"] <= TRAIN_END)]
    district_monthly_rates = compute_monthly_yoy_rates(
        _panel_for_rates, target_col, sid="overall", start_year=yoy_start_year
    )
    avg_annual_soc = np.mean([(1 + r) ** 12 - 1 for r in district_monthly_rates.values()])
    logger.info("  District YoY rates (used for society floor) — avg annual: %+.1f%%", avg_annual_soc * 100)

    vv_by_m = _vet_visits_by_month(panel)
    rows = []
    for _, origin in origin_rows.iterrows():
        sid = origin["series_id"]
        for h_idx, ym in enumerate(FORECAST_MONTHS[:24], 1):
            p = pd.Period(ym, freq="M"); tm = p.month
            cal = calendar_flags(p.year, tm); wx = wx_by_m.get(tm, {}); vv = vv_by_m.get(tm, {})
            si_row = seasonal_df[(seasonal_df["series_id"] == sid) & (seasonal_df["month_num"] == tm)]
            si_val = float(si_row["seasonal_index"].iloc[0]) if not si_row.empty else 1.0
            row = {"series_id": sid, "year_month": ym, "society": sid.replace("society:", ""),
                   "horizon": h_idx, "target_month_num": tm, "target_seasonal_index": si_val,
                   "series_id_cat": sid_to_cat.get(sid, 0),
                   "month_num": origin.get("month_num", 0)}
            for k in [f"lag_1{suffix}",f"lag_6{suffix}",f"lag_12{suffix}",
                       f"roll3{suffix}",f"roll6{suffix}",f"roll12{suffix}",
                       f"yoy_growth{suffix}",
                       "society_hist_mean","society_hist_std","district_share_lag1"]:
                row[k] = safe(origin.get(k, 0))
            # Cross-target lag features
            for k in ["lag_12_fat","rolling_3_mean_fat","lag_6_fat",
                      "lag_12","rolling_3_mean","lag_12_snf","rolling_3_mean_snf"]:
                if k in origin.index:
                    row[k] = safe(origin.get(k, 0))
            for c in ["avg_daily_milk","peak_day_milk","active_days","milk_concentration",
                       "avg_daily_fat","peak_day_fat","avg_daily_snf","peak_day_snf",
                       "unique_societies","total_transactions","cow_share",
                       "society_volume_tier_enc"]:
                if c in origin.index: row[c] = safe(origin[c])
            row.update(cal); row.update(wx); row.update(vv); rows.append(row)

    fdf = pd.DataFrame(rows)
    for c in feat_cols:
        if c not in fdf.columns: fdf[c] = 0
    X = fdf[feat_cols].fillna(0)
    p10_d, p50_d, p90_d = predict_triple((m10, m50, m90), X)
    si = fdf["target_seasonal_index"].values
    fdf["p10"] = np.round(np.maximum(p10_d * si, 0), 2)
    fdf["p50"] = np.round(np.maximum(p50_d * si, 0), 2)
    fdf["p90"] = np.round(np.maximum(p90_d * si, 0), 2)

    # Tactical YoY floor per society.
    # dead_band=5%: more tolerance than overall (society data is noisier).
    # max_scale=1.15: softer cap — don't force small/volatile societies too hard.
    # Societies with no last-year data (new societies) are left untouched by the floor.
    soc_hist = panel[panel["level"] == "society"][["series_id", "year_month", target_col]]
    result_pieces = []
    n_total_lifted = 0
    for sid, grp in fdf.groupby("series_id"):
        grp = grp.copy()
        soc_hist_sid = soc_hist[soc_hist["series_id"] == sid][["year_month", target_col]]
        if soc_hist_sid.empty or soc_hist_sid[target_col].sum() == 0:
            result_pieces.append(grp)
            continue
        before_count = n_total_lifted
        _year1_ref = grp[["year_month", "p50"]].rename(columns={"p50": target_col})
        # Exclude the partial last panel month from actuals — same reason as overall:
        # its actual is a partial-month value that would set an artificially low floor
        # for the same month next year. The forecast P50 from _year1_ref is used instead.
        soc_hist_sid_adj = soc_hist_sid[soc_hist_sid["year_month"] != FORECAST_MONTHS[0]]
        soc_hist_ext = pd.concat([soc_hist_sid_adj, _year1_ref], ignore_index=True).drop_duplicates(subset=["year_month"], keep="first")
        grp = _apply_yoy_floor(grp, soc_hist_ext, target_col, district_monthly_rates,
                               dead_band=0.0, max_scale=None, label=f"{sid}-pass1")
        # Pass 2: use floored year-1 P50 as anchor for year-2 months so
        # Jan-2028 is floored against Jan-2027's lifted value, not the raw prediction.
        _year1_ref_floored = grp[["year_month", "p50"]].rename(columns={"p50": target_col})
        soc_hist_ext2 = pd.concat([soc_hist_sid_adj, _year1_ref_floored], ignore_index=True).drop_duplicates(subset=["year_month"], keep="first")
        grp = _apply_yoy_floor(grp, soc_hist_ext2, target_col, district_monthly_rates,
                               dead_band=0.0, max_scale=None, label=f"{sid}-pass2")
        result_pieces.append(grp)
    fdf = pd.concat(result_pieces, ignore_index=True) if result_pieces else fdf

    n_out = fdf["series_id"].nunique()
    logger.info("Forecast produced for %s societies x %s months = %s rows",
                n_out, len(FORECAST_MONTHS[:24]), len(fdf))

    return fdf[["series_id","society","year_month","p10","p50","p90"]]

def collect_forecast_pieces(fc_overall, fc_society, target_col):
    """Return a list of DataFrames with standardised columns for later merge."""
    tname = TARGET_DISPLAY.get(target_col, target_col)
    pieces = []

    if not fc_overall.empty:
        keep = ["series_id", "year_month", "p10", "p50", "p90"]
        if "yoy_growth_pct" in fc_overall.columns:
            keep.append("yoy_growth_pct")
        ov = fc_overall[keep].copy()
        ov["level"] = "overall"
        ov["society"] = ""
        rename_map = {
            "p10": f"{target_col}_p10",
            "p50": f"{target_col}_p50",
            "p90": f"{target_col}_p90",
        }
        if "yoy_growth_pct" in ov.columns:
            rename_map["yoy_growth_pct"] = f"{target_col}_yoy_pct"
        ov.rename(columns=rename_map, inplace=True)
        pieces.append(ov)

    if not fc_society.empty:
        s = fc_society.copy()
        if "society" not in s.columns:
            s["society"] = s["series_id"].str.replace("society:", "", regex=False)
        s["level"] = "society"
        s = s[["series_id", "society", "year_month", "level", "p10", "p50", "p90"]]
        s.rename(columns={
            "p10": f"{target_col}_p10",
            "p50": f"{target_col}_p50",
            "p90": f"{target_col}_p90",
        }, inplace=True)
        pieces.append(s)

    logger.info("Collected %s forecast pieces: %s rows", tname, sum(len(p) for p in pieces))
    return pieces


def add_yoy_pct_columns(combined, panel):
    """Add {target}_yoy_pct = (p50 / same_month_last_year_actual - 1) × 100.

    For every forecast row, looks up the actual value from 12 months prior
    in the panel.  Example: April 2026 forecast p50 = 71 kg vs April 2025
    actual 67.83 kg → yoy_pct = +4.7%.
    """
    from config import MC_TARGETS as _TARGETS
    for target_col in _TARGETS:
        p50_col = f"{target_col}_p50"
        if p50_col not in combined.columns:
            continue
        if target_col not in panel.columns:
            continue
        # Build fast lookup: (series_id, year_month) → actual value
        hist = panel.loc[panel[target_col] > 0, ["series_id", "year_month", target_col]]
        lookup = dict(zip(zip(hist["series_id"], hist["year_month"]), hist[target_col]))

        # Compute last-year year_month for each forecast row (vectorised)
        ly_yms = combined["year_month"].map(
            lambda ym: (pd.Period(ym, freq="M") - 12).strftime("%Y-%m")
        )
        ly_vals = pd.array([
            lookup.get((sid, ly), None)
            for sid, ly in zip(combined["series_id"], ly_yms)
        ], dtype=object)
        ly_vals = pd.Series(ly_vals, index=combined.index, dtype="float64")

        yoy_col = f"{target_col}_yoy_pct"
        mask = ly_vals.notna() & (ly_vals > 0)
        combined[yoy_col] = None
        combined.loc[mask, yoy_col] = (
            (combined.loc[mask, p50_col] / ly_vals[mask] - 1) * 100
        ).round(2)
        logger.info("YoY %% (%s): computed for %s/%s rows", target_col, mask.sum(), len(combined))
    return combined


def build_milk_forecast_bundle(all_forecast_pieces: list, panel: pd.DataFrame) -> pd.DataFrame:
    """
    Combines all per-target forecast pieces into one row per (series_id, year_month).

    Pieces arrive as one DataFrame per (target × level), each carrying only that
    target's metric columns.  Concat stacks them as rows (one per target per month),
    then groupby.max() collapses to a single wide row with all metric columns filled.
    This avoids the _x/_y column suffix problem caused by merging DataFrames that
    share metric column names across overall and society pieces.
    """
    if not all_forecast_pieces:
        logger.warning("No forecast pieces to bundle.")
        return pd.DataFrame()

    merge_keys = ["series_id", "year_month", "level", "society"]

    frames = []
    for piece in all_forecast_pieces:
        if piece.empty:
            continue
        piece = piece.copy()
        if "society" in piece.columns:
            piece["society"] = piece["society"].fillna("").astype(str)
        frames.append(piece)

    if not frames:
        logger.warning("No forecast pieces to bundle (all empty).")
        return pd.DataFrame()

    # Stack all rows, then collapse to one row per key using max (ignores NaN)
    combined = pd.concat(frames, ignore_index=True)
    id_cols   = [c for c in merge_keys if c in combined.columns]
    metric_cols = [c for c in combined.columns if c not in id_cols]
    if metric_cols:
        combined = (combined.groupby(id_cols, as_index=False)[metric_cols]
                    .max()
                    .reset_index(drop=True))

    combined = combined.sort_values(["level", "series_id", "year_month"]).reset_index(drop=True)
    logger.info("Milk forecast bundle: %s rows  cols=%s", f"{len(combined):,}", list(combined.columns))
    return combined

# Others = Overall - sum(Top-N) — residual for consistency
# ══════════════════════════════════════════════════════════════════════════════

def compute_others_forecast(fc_overall, fc_society, panel=None, target_col="milk_kg"):
    """Compute Others forecast = overall_p50 - sum(society_p50) per month.
    If fc_overall is empty (fat/snf — no overall model), fall back to naive
    seasonal average from the Others series in the panel.
    """
    if fc_society.empty:
        return pd.DataFrame()

    if not fc_overall.empty:
        # Primary path: Others = Overall - sum(societies)
        soc_sum = fc_society.groupby("year_month")[["p10","p50","p90"]].sum().reset_index()
        ov = fc_overall[["year_month","p10","p50","p90"]].copy()
        merged = ov.merge(soc_sum, on="year_month", suffixes=("_ov","_soc"))
        others = pd.DataFrame({
            "series_id": OTHERS_SERIES_ID,
            "society": OTHERS_SOCIETY_NAME,
            "year_month": merged["year_month"],
            "p10": (merged["p10_ov"] - merged["p10_soc"]).clip(lower=0).round(2),
            "p50": (merged["p50_ov"] - merged["p50_soc"]).clip(lower=0).round(2),
            "p90": (merged["p90_ov"] - merged["p90_soc"]).clip(lower=0).round(2),
        })
        logger.info("Others forecast (%s): %s months, avg p50=%s  [residual method]",
                    target_col, len(others), f"{others['p50'].mean():,.0f}")
        return others

    # Fallback: naive seasonal average for Others (used for fat_kg, snf_kg)
    if panel is None or target_col not in panel.columns:
        return pd.DataFrame()
    oth_hist = panel[(panel["series_id"] == OTHERS_SERIES_ID) & (panel[target_col] > 0)].copy()
    if oth_hist.empty:
        return pd.DataFrame()
    oth_hist["month_num"] = pd.PeriodIndex(oth_hist["year_month"], freq="M").month
    monthly_avg = oth_hist.groupby("month_num")[target_col].mean()
    global_avg = oth_hist[target_col].mean()

    fc_months = sorted(fc_society["year_month"].unique())
    rows = []
    for ym in fc_months:
        mn = pd.Period(ym, freq="M").month
        val = round(float(monthly_avg.get(mn, global_avg)), 2)
        rows.append({"series_id": OTHERS_SERIES_ID, "society": OTHERS_SOCIETY_NAME,
                      "year_month": ym, "p10": round(val * 0.85, 2),
                      "p50": val, "p90": round(val * 1.15, 2)})
    others = pd.DataFrame(rows)
    logger.info("Others forecast (%s): %s months, avg p50=%s  [naive seasonal method]",
                target_col, len(others), f"{others['p50'].mean():,.0f}")
    return others


def compute_others_test(ov_test, soc_test, target_col, panel=None):
    """Compute Others test predictions = overall_test - sum(society_test) per month.
    If ov_test is empty (fat/snf — no overall model), compute from panel actuals.
    """
    if soc_test.empty:
        return pd.DataFrame()

    if not ov_test.empty:
        soc_sum = soc_test.groupby("year_month")[[target_col,"p10","p50","p90"]].sum().reset_index()
        merged = ov_test[["year_month",target_col,"p10","p50","p90"]].merge(
            soc_sum, on="year_month", suffixes=("_ov","_soc"))
        others = pd.DataFrame({
            "series_id": OTHERS_SERIES_ID,
            "year_month": merged["year_month"],
            target_col: (merged[f"{target_col}_ov"] - merged[f"{target_col}_soc"]).clip(lower=0).round(2),
            "p10": (merged["p10_ov"] - merged["p10_soc"]).clip(lower=0).round(2),
            "p50": (merged["p50_ov"] - merged["p50_soc"]).clip(lower=0).round(2),
            "p90": (merged["p90_ov"] - merged["p90_soc"]).clip(lower=0).round(2),
        })
        logger.info("Others test (%s): %s months  [residual method]", target_col, len(others))
        return others

    # Fallback: use actual Others from panel + naive prediction
    if panel is None or target_col not in panel.columns:
        return pd.DataFrame()
    test_months = sorted(soc_test["year_month"].unique())
    oth_actual = panel[(panel["series_id"] == OTHERS_SERIES_ID) &
                       (panel["year_month"].isin(test_months))][["year_month", target_col]].copy()
    if oth_actual.empty:
        return pd.DataFrame()
    # Naive prediction: use month-of-year average from historical Others
    oth_hist = panel[(panel["series_id"] == OTHERS_SERIES_ID)].copy()
    oth_hist["month_num"] = pd.PeriodIndex(oth_hist["year_month"], freq="M").month
    monthly_avg = oth_hist.groupby("month_num")[target_col].mean()
    global_avg = oth_hist[target_col].mean()

    rows = []
    for _, r in oth_actual.iterrows():
        mn = pd.Period(r["year_month"], freq="M").month
        val = round(float(monthly_avg.get(mn, global_avg)), 2)
        rows.append({"series_id": OTHERS_SERIES_ID, "year_month": r["year_month"],
                      target_col: round(float(r[target_col]), 2),
                      "p10": round(val * 0.85, 2), "p50": val, "p90": round(val * 1.15, 2)})
    others = pd.DataFrame(rows)
    logger.info("Others test (%s): %s months  [naive seasonal method]", target_col, len(others))
    return others

# ══════════════════════════════════════════════════════════════════════════════
# YoY Growth Monitor
# ══════════════════════════════════════════════════════════════════════════════

def _monitor_yoy_growth(panel: pd.DataFrame, combined_fc: pd.DataFrame, out_dir):
    """
    Compute and print Year-over-Year growth for milk_kg across all available
    years (actuals + forecast).
    """
    logger.info("YoY GROWTH MONITOR — Milk Collection (milk_kg)")

    # ── Actuals: overall series from panel ───────────────────────────
    ov_actual = panel[panel["series_id"] == "overall"].copy()
    if "milk_kg" not in ov_actual.columns or ov_actual.empty:
        logger.warning("No overall milk_kg actuals in panel — skipping YoY check.")
        return

    ov_actual["year"] = pd.PeriodIndex(ov_actual["year_month"], freq="M").year
    act_raw    = ov_actual.groupby("year")["milk_kg"].sum().to_dict()
    act_counts = ov_actual.groupby("year")["year_month"].count().to_dict()
    # Annualize partial years so comparisons are always on a full-year basis
    act_by_year = {yr: int(round(v * 12 / act_counts[yr])) for yr, v in act_raw.items()}

    # ── Forecast: overall rows from combined forecast DataFrame ──────────────────
    fc_by_year: dict = {}
    fc_counts:  dict = {}
    if not combined_fc.empty and "milk_kg_p50" in combined_fc.columns and "series_id" in combined_fc.columns:
        ov_fc = combined_fc[combined_fc["series_id"] == "overall"].copy()
        if not ov_fc.empty:
            ov_fc["year"] = ov_fc["year_month"].str[:4].astype(int)
            fc_raw    = ov_fc.groupby("year")["milk_kg_p50"].sum().to_dict()
            fc_counts = ov_fc.groupby("year")["year_month"].count().to_dict()
            fc_by_year = {yr: int(round(v * 12 / fc_counts[yr])) for yr, v in fc_raw.items()}

    all_years   = sorted(set(act_by_year.keys()) | set(fc_by_year.keys()))
    yoy_records = []
    all_pass    = True

    for yr in all_years:
        actual_val   = act_by_year.get(yr)
        forecast_val = fc_by_year.get(yr)
        source       = "actual" if yr in act_by_year else "forecast"
        this_val     = actual_val if actual_val is not None else forecast_val
        prev_val     = act_by_year.get(yr - 1) or fc_by_year.get(yr - 1)
        n_months     = act_counts.get(yr) or fc_counts.get(yr) or 12

        yoy_pct = None
        status  = "—"
        if this_val and prev_val and prev_val > 0:
            yoy_pct = round((this_val - prev_val) / prev_val * 100, 1)
            status  = (
                "[OK]  within ±20%" if abs(yoy_pct) <= 20 else
                "[WARN] 20-40%"     if abs(yoy_pct) <= 40 else
                "[!]   >40%"
            )
            if abs(yoy_pct) > 20 and source == "forecast":
                all_pass = False

        yoy_records.append({
            "year": yr, "source": source, "n_months": n_months,
            "annual_total_milk_kg": int(this_val) if this_val else None,
            "yoy_pct": yoy_pct, "status": status,
        })

    # Log table — annotate partial years with "(N mo)" so it is clear they were annualized
    logger.info("%-6s %-16s %16s %9s  Status", "Year", "Source", "Ann. milk_kg", "YoY %")
    for r in yoy_records:
        total_s  = f"{r['annual_total_milk_kg']:>16,}" if r['annual_total_milk_kg'] else f"{'—':>16}"
        yoy_s    = f"{r['yoy_pct']:>+8.1f}%" if r['yoy_pct'] is not None else f"{'—':>9}"
        src_note = f"{r['source']} ({r['n_months']} mo)" if r['n_months'] < 12 else r['source']
        log_fn   = logger.info if r['yoy_pct'] is None or abs(r['yoy_pct']) <= 20 else logger.warning
        log_fn("%-6s %-16s %s %s  %s", r['year'], src_note, total_s, yoy_s, r['status'])

    if all_pass:
        logger.info("Overall YoY check: PASS")
    else:
        logger.warning("Overall YoY check: SOME forecasts >20%% YoY — review recommended")


def main():
    span = get_current_span()
    global _VER_TAG
    ver_num, _VER_TAG = _read_current_version(MODEL_DIR)
    logger.info("Milk model version for this run: %s", _VER_TAG)
    logger.info("Step 6 — Milk Collection Training")
    logger.info("TRAIN -> %s  TEST %s->%s  FC %s->%s",
                TRAIN_END, TEST_START, TEST_END, FORECAST_MONTHS[0], FORECAST_MONTHS[-1])
    logger.info("6 Models: Overall(milk/fat/snf) x 3 + Society(milk/fat/snf) x 3")

    OUT_DIR.mkdir(parents=True, exist_ok=True)

    span.add_event("read_panel_start")
    t0 = time.time()
    panel = pd.read_parquet(PANEL_PATH)
    span.add_event("read_panel_done", {"rows": len(panel), "series": panel["series_id"].nunique(), "elapsed_s": round(time.time() - t0, 2)})
    logger.info("Panel: %s rows  %s series  %s cols",
                f"{len(panel):,}", panel['series_id'].nunique(), len(panel.columns))
    has_others = OTHERS_SERIES_ID in panel["series_id"].values
    if has_others:
        logger.info("Others residual series detected — will compute __OTHERS__ forecasts")
    else:
        logger.info("No Others residual — all societies modelled individually")

    for level in ["overall", "society"]:
        ldf = panel[panel["level"] == level]
        if not ldf.empty:
            months = sorted(ldf["year_month"].unique())
            logger.info("%s: %s -> %s (%s months)", level, months[0], months[-1], len(months))

    all_eval = []
    all_test_pieces = []
    all_forecast_pieces = []          # ← collect for unified CSV

    for t_idx, target_col in enumerate(MC_TARGETS, 1):
        if target_col not in panel.columns:
            logger.warning("[SKIP] %s not in panel — skipping", target_col)
            continue

        tname = TARGET_DISPLAY.get(target_col, target_col)
        span.add_event("target_loop_start", {"target": target_col})
        t_target = time.time()
        logger.info("=" * 60)
        logger.info("TARGET %d/%d — %s", t_idx, len(MC_TARGETS), tname)
        logger.info("=" * 60)

        # ── Model A — Overall (all targets: milk_kg, fat_kg, snf_kg) ─────
        logger.info("--- Model A: Overall ---")
        ov_models = None; ov_feat = []; ov_df = pd.DataFrame()
        ov_eval = []; ov_test = pd.DataFrame(); ov_seasonal = pd.DataFrame()

        ov_result = build_overall_model(panel, target_col)
        ov_models, ov_feat, ov_df, ov_eval, ov_test, ov_seasonal = ov_result
        logger.info("--- Model A done ---")

        # ── Model B — Society (ALL targets) ───────────────────────────────
        logger.info("--- Model B: Society ---")
        soc_result = build_society_model(panel, target_col)
        soc_models, soc_feat, soc_df, sid_map, soc_eval, soc_test, soc_seasonal = soc_result
        logger.info("--- Model B done ---")

        # Save test predictions
        test_pieces = []
        if not ov_test.empty:
            t = ov_test.copy(); t["model"] = f"A_overall"; t["target"] = target_col
            test_pieces.append(t[["model","target","series_id","year_month",target_col,"p10","p50","p90"]])
        if not soc_test.empty:
            t = soc_test.copy(); t["model"] = f"B_society"; t["target"] = target_col
            test_pieces.append(t[["model","target","series_id","year_month",target_col,"p10","p50","p90"]])

        # ── Others test predictions (only when __OTHERS__ residual series exists) ──
        if has_others:
            others_test = compute_others_test(ov_test, soc_test, target_col, panel=panel)
            if not others_test.empty:
                ot = others_test.copy(); ot["model"] = "B_society"; ot["target"] = target_col
                test_pieces.append(ot[["model","target","series_id","year_month",target_col,"p10","p50","p90"]])

        if test_pieces:
            tdf = pd.concat(test_pieces, ignore_index=True)
            # No CSV saved — test predictions kept in memory only
            all_test_pieces.append(tdf)

        # Forecast
        logger.info("--- Forecasting %s ---", tname)
        span.add_event("forecast_start", {"target": target_col})
        t_fc = time.time()
        fc_overall = pd.DataFrame()
        if ov_models is not None:
            fc_overall = forecast_overall(ov_models, ov_feat, ov_df, panel, ov_seasonal, target_col)
        fc_society = pd.DataFrame()
        if soc_models is not None:
            fc_society = forecast_society(soc_models, soc_feat, soc_df, sid_map, panel, soc_seasonal, target_col)
        elapsed_fc = round(time.time() - t_fc, 1)
        span.add_event("forecast_done", {"target": target_col, "overall_rows": len(fc_overall), "society_rows": len(fc_society), "elapsed_s": elapsed_fc})
        logger.info("--- Forecast done  overall=%s rows  society=%s rows  (%.1fs) ---",
                    len(fc_overall), len(fc_society), elapsed_fc)

        # ── Others forecast (only when __OTHERS__ residual series exists) ──
        if has_others:
            others_fc = compute_others_forecast(fc_overall, fc_society, panel=panel, target_col=target_col)
            if not others_fc.empty:
                fc_society = pd.concat([fc_society, others_fc], ignore_index=True)

        # Collect forecast pieces for the unified CSV
        pieces = collect_forecast_pieces(fc_overall, fc_society, target_col)
        all_forecast_pieces.extend(pieces)

        # Collect eval
        for e in ov_eval + soc_eval:
            e["target"] = target_col
        all_eval.extend(ov_eval + soc_eval)
        elapsed_target = round(time.time() - t_target, 1)
        span.add_event("target_loop_done", {"target": target_col, "elapsed_s": elapsed_target})
        logger.info("TARGET %d/%d — %s complete  (%.1fs)", t_idx, len(MC_TARGETS), tname, elapsed_target)

    # ── Assemble forecast bundle DataFrame ────────────────────────────────
    combined_fc = build_milk_forecast_bundle(all_forecast_pieces, panel)

    # ── Save forecast CSV (unified, all targets, all levels) ──────────────
    if combined_fc is not None and not combined_fc.empty:
        MC_FORECAST_CSV.parent.mkdir(parents=True, exist_ok=True)
        # Re-compute YoY % columns against panel actuals before writing
        combined_fc = add_yoy_pct_columns(combined_fc, panel)

        combined_fc.to_csv(MC_FORECAST_CSV, index=False)
        logger.info("Saved forecast CSV: %s  (%s rows)", MC_FORECAST_CSV, f"{len(combined_fc):,}")

    # ── Save test-predictions CSV (all targets in one file) ───────────────
    if all_test_pieces:
        combined_test = pd.concat(all_test_pieces, ignore_index=True)
        test_csv = OUT_DIR / "test_predictions.csv"
        combined_test.to_csv(test_csv, index=False)
        logger.info("Saved test predictions: %s  (%s rows)", test_csv, f"{len(combined_test):,}")

    # ── Save evaluation CSV ───────────────────────────────────────────────
    if all_eval:
        eval_df = pd.DataFrame(all_eval)
        eval_csv = OUT_DIR / "evaluation.csv"
        eval_df.to_csv(eval_csv, index=False)
        logger.info("Saved evaluation: %s  (%s rows)", eval_csv, f"{len(eval_df):,}")

    # ── YoY Growth Monitor ────────────────────────────────────────────────────
    _monitor_yoy_growth(panel, combined_fc if combined_fc is not None else pd.DataFrame(), OUT_DIR)

    # ── Save forecast bundle PKL ──────────────────────────────────────────────
    span.add_event("save_bundle_start")
    t_save = time.time()
    logger.info("Saving milk forecast bundle and versioning models...")
    if combined_fc is not None and not combined_fc.empty:
        bundle_path = MODEL_DIR / f"{_VER_TAG}_forecast_bundle.pkl"
        with open(bundle_path, "wb") as f:
            pickle.dump({
                "forecasts":    combined_fc.to_dict(orient="records"),
                "version":      _VER_TAG,
                "trained_at":   datetime.utcnow().isoformat(),
                "domain":       "milk_collection",
            }, f)
        logger.info("Saved forecast bundle: %s", bundle_path)

        # Upload to MinIO for durable storage
        try:
            from pipeline.common.storage import upload_file
            upload_file(bundle_path, "milk_collection", f"{_VER_TAG}_forecast_bundle.pkl")
            upload_file(MC_FORECAST_CSV, "milk_collection", "all_forecasts.csv")
        except ImportError:
            logger.warning("pipeline.common.storage not available — skipping MinIO upload")

    # ── Write version pointer ─────────────────────────────────────────────────
    (MODEL_DIR / "current_version.json").write_text(json.dumps({
        "version": ver_num, "tag": _VER_TAG,
        "trained_at": datetime.utcnow().isoformat()
    }, indent=2))
    logger.info("Written: current_version.json  (tag=%s)", _VER_TAG)

    # ── Delete previous version PKLs ─────────────────────────────────────────
    # Use f"{_VER_TAG}_" prefix (with underscore) to avoid "v1" matching "v10"
    deleted = []
    for old in MODEL_DIR.glob("v*_lgbm*.pkl"):
        if not old.name.startswith(f"{_VER_TAG}_"):
            old.unlink(missing_ok=True)
            deleted.append(old.name)
    for old in MODEL_DIR.glob("v*_forecast_bundle.pkl"):
        if not old.name.startswith(f"{_VER_TAG}_"):
            old.unlink(missing_ok=True)
            deleted.append(old.name)
    if deleted:
        logger.info("Deleted old version files: %s", deleted)

    span.add_event("save_bundle_done", {"elapsed_s": round(time.time() - t_save, 2)})

    logger.info("DONE — 6 models trained:")
    logger.info("  1. Overall  milk_kg  (district-level)")
    logger.info("  2. Overall  fat_kg   (district-level)")
    logger.info("  3. Overall  snf_kg   (district-level)")
    logger.info("  4. Society  milk_kg  (all societies)")
    logger.info("  5. Society  fat_kg   (all societies)")
    logger.info("  6. Society  snf_kg   (all societies)")
    logger.info("  Forecast bundle: %s_forecast_bundle.pkl", _VER_TAG)
    span.add_event("step06_complete", {"version": _VER_TAG, "total_elapsed_s": round(time.time() - t0, 2)})


if __name__ == "__main__":
    main()
